#### Paso 3.2: Verificar KYC del Inversor

**Script**: `scripts/demo/set_kyc.js`

```javascript
// scripts/demo/set_kyc.js
const { ethers } = require("hardhat");

async function main() {
    const complianceAddress = "0x9Ea652E99B9D488e53e418726642d872824c75D4";
    const compliance = await ethers.getContractAt("ComplianceManager", complianceAddress);
    
    // Dirección del inversor a verificar
    const investorAddress = "0x[INVESTOR_ADDRESS]"; // Reemplazar
    
    // Verificar KYC (2 = Verified)
    const tx = await compliance.setKYCStatus(investorAddress, 2);
    await tx.wait();
    
    console.log(`✅ KYC Verificado para: ${investorAddress}`);
    
    // Verificar estado
    const status = await compliance.kycStatus(investorAddress);
    console.log("Estado KYC:", status); // 2 = Verified
}

main().catch(console.error);
```

**Ejecutar**:
```bash
npx hardhat run scripts/demo/set_kyc.js --network baseSepolia
```

#### Paso 3.4: Comprar Tokens

**Script**: `scripts/demo/purchase_tokens.js`

```javascript
// scripts/demo/purchase_tokens.js
const { ethers } = require("hardhat");

async function main() {
    // Usar la wallet del inversor
    const [, investor] = await ethers.getSigners();
    
    const complianceAddress = "0x9Ea652E99B9D488e53e418726642d872824c75D4";
    const usdcAddress = "0x70Fef40966f45fA2cdf2c241c247D806B9E24d61";
    
    const compliance = await ethers.getContractAt("ComplianceManager", complianceAddress);
    const usdc = await ethers.getContractAt("MockERC20", usdcAddress);
    
    const franchiseId = 1;
    const tokensToBuy = 100; // Comprar 100 tokens
    const paymentAmount = ethers.parseUnits("1000", 18); // $1,000 (10 USD por token)
    
    // Aprobar USDC
    console.log("Aprobando USDC...");
    let tx = await usdc.connect(investor).approve(complianceAddress, paymentAmount);
    await tx.wait();
    
    // Comprar tokens
    console.log("Comprando tokens...");
    tx = await compliance.connect(investor).purchaseTokens(franchiseId, tokensToBuy, paymentAmount);
    const receipt = await tx.wait();
    
    console.log(`✅ Compra exitosa!`);
    console.log(`TX Hash: ${receipt.hash}`);
    console.log(`BaseScan: https://sepolia.basescan.org/tx/${receipt.hash}`);
}

main().catch(console.error);
```

**Talking Point**:
"El inversor aprueba el pago en USDC y compra 100 tokens por $1,000. El ComplianceManager verifica el KYC automáticamente y mintea los tokens"

### **Parte 4: Distribución de Dividendos** (5 min)

#### Paso 4.1: Depositar Dividendos

**Script**: `scripts/demo/deposit_dividends.js`

```javascript
// scripts/demo/deposit_dividends.js
const { ethers } = require("hardhat");

async function main() {
    const distributorAddress = "0xC8e3575e66FB54d0B00237f9b17dc6e05BF8d4B1";
    const usdcAddress = "0x70Fef40966f45fA2cdf2c241c247D806B9E24d61";
    
    const distributor = await ethers.getContractAt("DividendDistributor", distributorAddress);
    const usdc = await ethers.getContractAt("MockERC20", usdcAddress);
    
    const franchiseId = 1;
    const dividendAmount = ethers.parseUnits("5000", 18); // $5,000 en dividendos
    
    // Mintear USDC al manager para dividendos
    console.log("Minteando USDC para dividendos...");
    let tx = await usdc.mint(await ethers.provider.getSigner().getAddress(), dividendAmount);
    await tx.wait();
    
    // Aprobar
    console.log("Aprobando USDC...");
    tx = await usdc.approve(distributorAddress, dividendAmount);
    await tx.wait();
    
    // Depositar dividendos
    console.log("Depositando dividendos...");
    tx = await distributor.depositDividends(franchiseId, dividendAmount);
    const receipt = await tx.wait();
    
    console.log(`✅ Dividendos depositados: $5,000`);
    console.log(`TX Hash: ${receipt.hash}`);
}

main().catch(console.error);
```

**Talking Point**:
"El manager de la franquicia deposita $5,000 en ganancias para distribuir a los holders"

#### Paso 4.2: Crear Ciclo de Dividendos (Manual)

**Script**: `scripts/demo/trigger_dividend_cycle.js`

```javascript
// scripts/demo/trigger_dividend_cycle.js
const { ethers } = require("hardhat");

async function main() {
    const distributorAddress = "0xC8e3575e66FB54d0B00237f9b17dc6e05BF8d4B1";
    const distributor = await ethers.getContractAt("DividendDistributor", distributorAddress);
    
    const franchiseId = 1;
    const performData = ethers.AbiCoder.defaultAbiCoder().encode(['uint256'], [franchiseId]);
    
    console.log("Creando ciclo de dividendos...");
    const tx = await distributor.performUpkeep(performData);
    const receipt = await tx.wait();
    
    const cycleId = await distributor.currentCycleId(franchiseId);
    console.log(`✅ Ciclo creado: #${cycleId}`);
    console.log(`TX Hash: ${receipt.hash}`);
}

main().catch(console.error);
```

**Talking Point**:
"En producción, Chainlink Automation ejecutaría esto automáticamente cada hora. Para la demo, lo hacemos manualmente"
